## 3. VIP冻结和母体

实际使用的核心CTE：

```sql
WITH login_snapshot AS (
  SELECT "#user_id" uid,
    MAX(CASE WHEN "#event_time">=TIMESTAMP '2026-08-24 09:00:00'
              AND "#event_time"< TIMESTAMP '2026-08-27 09:00:00' THEN 1 ELSE 0 END) has_main_login,
    max_by(viplevel,"#event_time") FILTER (WHERE "#event_time"<TIMESTAMP '2026-08-24 09:00:00' AND viplevel IS NOT NULL) pre_vip,
    max_by(level,"#event_time") FILTER (WHERE "#event_time"<TIMESTAMP '2026-08-24 09:00:00' AND viplevel IS NOT NULL) pre_level,
    max_by(chip,"#event_time") FILTER (WHERE "#event_time"<TIMESTAMP '2026-08-24 09:00:00' AND viplevel IS NOT NULL) pre_chip,
    max_by(paycount,"#event_time") FILTER (WHERE "#event_time"<TIMESTAMP '2026-08-24 09:00:00' AND viplevel IS NOT NULL) pre_paycount,
    max_by(totalpay,"#event_time") FILTER (WHERE "#event_time"<TIMESTAMP '2026-08-24 09:00:00' AND viplevel IS NOT NULL) pre_totalpay,
    min_by(viplevel,"#event_time") FILTER (WHERE "#event_time">=TIMESTAMP '2026-08-24 09:00:00' AND "#event_time"<TIMESTAMP '2026-08-27 09:00:00' AND viplevel IS NOT NULL) post_vip,
    min_by(level,"#event_time") FILTER (WHERE "#event_time">=TIMESTAMP '2026-08-24 09:00:00' AND "#event_time"<TIMESTAMP '2026-08-27 09:00:00' AND viplevel IS NOT NULL) post_level,
    min_by(chip,"#event_time") FILTER (WHERE "#event_time">=TIMESTAMP '2026-08-24 09:00:00' AND "#event_time"<TIMESTAMP '2026-08-27 09:00:00' AND viplevel IS NOT NULL) post_chip
  FROM v_event_61
  WHERE "$part_date" BETWEEN '2026-07-25' AND '2026-08-27' AND "#event_name"='event_1000'
  GROUP BY 1
), snapshot AS (
  SELECT uid,COALESCE(pre_vip,post_vip,0) vip,COALESCE(pre_level,post_level,0) start_level,
         COALESCE(pre_chip,post_chip,0) start_chip,
         CASE WHEN pre_vip IS NOT NULL THEN 'pre_start'
              WHEN post_vip IS NOT NULL THEN 'fallback_after_start' ELSE 'missing' END snapshot_source
  FROM login_snapshot WHERE has_main_login=1
)
SELECT CASE WHEN vip=0 THEN 'VIP0' WHEN vip=1 THEN 'VIP1' WHEN vip=2 THEN 'VIP2' ELSE 'VIP3+' END vip_group,
       snapshot_source,COUNT(*) users
FROM snapshot GROUP BY 1,2;
```

结果：VIP0/1/2/3+为1,723/317/18/40；fallback为1,197/55/0/1。对应报告第3节。

## 4. 漏斗查询

每个步骤先按用户去重，再回连母体；严格列使用累计 `CASE WHEN`。活动参数分别替换1033/1034。

```sql
WITH <snapshot>,
init AS (SELECT "#user_id" uid FROM v_event_61 WHERE "$part_date" BETWEEN '2026-08-24' AND '2026-08-27' AND "#event_name"='event_20000' AND activity_id=:activity_id AND reason='252' AND "#event_time">=:start AND "#event_time"<:end GROUP BY 1),
exposure AS (SELECT "#user_id" uid FROM v_event_61 WHERE "$part_date" BETWEEN '2026-08-24' AND '2026-08-27' AND "#event_name"='ta_open_panel' AND LOWER(panel_name) IN (:exposure_enum) AND "#event_time">=:start AND "#event_time"<:end GROUP BY 1),
opened AS (SELECT "#user_id" uid FROM v_event_61 WHERE "$part_date" BETWEEN '2026-08-24' AND '2026-08-27' AND "#event_name"='ta_open_panel' AND LOWER(panel_name) IN (:open_enum) AND "#event_time">=:start AND "#event_time"<:end GROUP BY 1),
slots AS (SELECT "#user_id" uid FROM v_event_61 WHERE "$part_date" BETWEEN '2026-08-24' AND '2026-08-27' AND "#event_name"='event_7179' AND "#event_time">=:start AND "#event_time"<:end GROUP BY 1),
points AS (SELECT "#user_id" uid FROM v_event_61 WHERE "$part_date" BETWEEN '2026-08-24' AND '2026-08-27' AND "#event_name"='event_10074' AND activity_id=:activity_id AND reason='SlotMachineBet' AND "#event_time">=:start AND "#event_time"<:end GROUP BY 1),
item AS (SELECT "#user_id" uid FROM v_event_61 WHERE "$part_date" BETWEEN '2026-08-24' AND '2026-08-27' AND "#event_name"='event_20000' AND activity_id=:activity_id AND reason='251' AND "#event_time">=:start AND "#event_time"<:end GROUP BY 1),
used AS (SELECT "#user_id" uid FROM v_event_61 WHERE "$part_date" BETWEEN '2026-08-24' AND '2026-08-27' AND "#event_name"='event_20000' AND activity_id=:activity_id AND reason='254' AND "#event_time">=:start AND "#event_time"<:end GROUP BY 1),
reward AS (SELECT "#user_id" uid FROM v_event_61 WHERE "$part_date" BETWEEN '2026-08-24' AND '2026-08-27' AND "#event_name"='event_20001' AND activity_id=:activity_id AND "#event_time">=:start AND "#event_time"<:end GROUP BY 1)
SELECT <vip_group>,COUNT(*) eligible_users,COUNT(i.uid) init_users,COUNT(ex.uid) exposure_users,
       COUNT(o.uid) open_users,COUNT(sl.uid) slots_users,COUNT(p.uid) point_users,COUNT(it.uid) item_users,
       COUNT(u.uid) use_users,COUNT(r.uid) reward_users,
       SUM(CASE WHEN i.uid IS NOT NULL AND ex.uid IS NOT NULL AND o.uid IS NOT NULL AND sl.uid IS NOT NULL
                 AND p.uid IS NOT NULL AND it.uid IS NOT NULL AND u.uid IS NOT NULL AND r.uid IS NOT NULL THEN 1 ELSE 0 END) strict_reward
FROM snapshot s LEFT JOIN init i ON s.uid=i.uid LEFT JOIN exposure ex ON s.uid=ex.uid
LEFT JOIN opened o ON s.uid=o.uid LEFT JOIN slots sl ON s.uid=sl.uid LEFT JOIN points p ON s.uid=p.uid
LEFT JOIN item it ON s.uid=it.uid LEFT JOIN used u ON s.uid=u.uid LEFT JOIN reward r ON s.uid=r.uid
GROUP BY 1;
```

结果映射：报告第4节。另以用户CTE汇总事件量，得到各VIP积分、道具、使用和奖励的人均量。

## 5. 互斥流失归因

用户级flags只存在SQL CTE内，最终按VIP/原因聚合。分类顺序固定，保证与未获关键奖励母体对账：

```sql
CASE
 WHEN has_exposure=0 THEN '02_登录但无活动曝光'
 WHEN has_open=0 THEN '03_曝光但未打开'
 WHEN has_slots=0 THEN '04_打开但无Slots'
 WHEN has_points=0 THEN '05_有Slots但无有效进度'
 WHEN has_item=0 THEN '06_有积分但未获得道具'
 WHEN has_use=0 THEN '07_获得道具但未使用'
 WHEN low_balance=0 THEN '08_已使用但未到下一里程碑'
 WHEN low_balance=1 THEN '09_持金不足以到下一里程碑'
 WHEN broken_control=1 AND has_product=0 THEN '10_破产但无商品曝光'
 WHEN has_product=1 AND has_request=0 THEN '11_商品曝光但未发起购买'
 WHEN has_request=1 AND has_success=0 THEN '12_发起购买但失败'
 ELSE '13_暂时无法分类' END
```

`low_balance=MAX(costbet>0 AND start_value<3*costbet)`；`broken_control=MAX(controltypeint=3)`。薯片未奖励母体1,653人，777为1,763人，分类合计完全一致。对应报告第5节。

## 6. RTP P0验证

字段与重复验证：

```sql
SELECT COUNT(*) spins,
 SUM(CASE WHEN winchip<0 THEN 1 ELSE 0 END) negative_win,
 SUM(CASE WHEN winchip=0 THEN 1 ELSE 0 END) zero_win,
 SUM(CASE WHEN winchip>0 THEN 1 ELSE 0 END) positive_win,
 SUM(costbet) costbet_sum,SUM(winchip) winchip_sum
FROM v_event_61
WHERE "$part_date" BETWEEN '2026-08-24' AND '2026-08-27' AND "#event_name"='event_7179'
AND "#event_time">=TIMESTAMP '2026-08-24 09:00:00' AND "#event_time"<TIMESTAMP '2026-08-27 09:00:00';
```

结果：1,691,629条；负/零/正为0/992,760/698,869。免费局按 `isfreespin,rtpswitchid` 单独汇总；完整签名去重后精确重复0。

美元归一化SQL结构：

```sql
WITH prices(level_band,vip,coins_per_usd) AS (VALUES :price_matrix_336_rows),
spins AS (
 SELECT "#user_id" uid,costbet,winchip,isfreespin,
        :nearest_lower_level_band AS level_band,
        LEAST(GREATEST(COALESCE(viplevel,0),0),15) vip
 FROM v_event_61
 WHERE "$part_date" BETWEEN '2026-08-24' AND '2026-08-27' AND "#event_name"='event_7179'
 AND "#event_time">=TIMESTAMP '2026-08-24 09:00:00' AND "#event_time"<TIMESTAMP '2026-08-27 09:00:00'
)
SELECT COUNT(*) spins,COUNT(DISTINCT uid) users,
 SUM(costbet/coins_per_usd) gross_usd,SUM(winchip/coins_per_usd) payout_usd,
 SUM((costbet-winchip)/coins_per_usd) net_usd,
 SUM(CASE WHEN isfreespin=1 THEN winchip/coins_per_usd ELSE 0 END) free_payout_usd
FROM spins JOIN prices USING(level_band,vip);
```

结果：$734,867.44/$679,595.46/$55,271.98/$172,105.99，对应报告第6节。

## 7. RTP档位匹配

结构化参数：`threshold_matrix_336_rows` 来自 `SlotsCasinoResultSwitchStrategy` 按 `(level,vip)` 去重；机器白名单33台；Bet档来自事件 `betlevel`。

```sql
WITH thresholds(level_band,runtime_vip,threshold_betindex) AS (VALUES :threshold_matrix_336_rows),
prices(level_band,runtime_vip,coins_per_usd) AS (VALUES :price_matrix_336_rows),
spins AS (<主窗口event_7179并计算level_band>),
classified AS (
 SELECT sp.*,t.threshold_betindex,p.coins_per_usd,
 CASE
  WHEN machineid NOT IN (:machine_ids_33) OR t.threshold_betindex IS NULL OR betlevel IS NULL THEN 'unmapped'
  WHEN rtpswitchid IN (3,4) THEN 'actual105/120'
  WHEN rtpswitchid NOT IN (1,2) THEN 'unmapped'
  WHEN betlevel<threshold_betindex AND rtpswitchid=1 THEN '95/95'
  WHEN betlevel<threshold_betindex AND rtpswitchid=2 THEN '95/85'
  WHEN betlevel>=threshold_betindex AND rtpswitchid=2 THEN '85/85'
  WHEN betlevel>=threshold_betindex AND rtpswitchid=1 THEN '85/95'
  ELSE 'unmapped' END match_class
 FROM spins sp LEFT JOIN thresholds t USING(level_band,runtime_vip)
 LEFT JOIN prices p USING(level_band,runtime_vip)
)
SELECT :dimension,match_class,COUNT(*) spins,COUNT(DISTINCT uid) users,
 SUM(costbet/coins_per_usd) gross_usd,SUM(winchip/coins_per_usd) payout_usd
FROM classified GROUP BY 1,2;
```

`:dimension` 分别实际执行：冻结VIP、Level band、Machine、Bet档、UTC日期、`chips_depth|lucky_depth`。总结果：95/95 1,333,062；95/85 598；85/85 11,983；85/95 16,802；105/120 329,138；unmapped 40。对应报告第7节。

## 8. 奖励美元价值

event_20001 与 event_1011 同用户、3秒内最近事件匹配；19,340条QuestJackpotReward全部匹配，18,922条同毫秒、418条1秒内。最终仅聚合：

```sql
WITH activity_reward AS (<event_20001 activity_id in 1033,1034>),
coin_reward AS (<event_1011 reason='QuestJackpotReward'>),
matched AS (
 SELECT c.uid,c.event_time,c.change,c.chip,c.chip_dollar,
        min_by(a.activity_id,ABS(date_diff('millisecond',a.event_time,c.event_time))) activity_id,
        MIN(ABS(date_diff('millisecond',a.event_time,c.event_time))) delta_ms
 FROM coin_reward c LEFT JOIN activity_reward a
 ON c.uid=a.uid AND ABS(date_diff('millisecond',a.event_time,c.event_time))<=3000
 GROUP BY 1,2,3,4,5
)
SELECT activity_id,COUNT(*) events,COUNT(DISTINCT uid) users,SUM(change) change_sum,
 SUM(CASE WHEN chip>0 AND chip_dollar IS NOT NULL THEN change*chip_dollar/chip/100.0 ELSE 0 END) approx_usd
FROM matched GROUP BY 1;
```

结果：1033约$8,275.85；1034约$375.33。再按event_1011 reason汇总，SnackBoxReward约$8.79、QuestPassReward约$9.66、StrikeLuckySpinCoin约$1,089.47。对应报告第2、9、10节。公式为近似现金等价，不含卡包等非现金项。

## 11. 查询—结论映射与复现规则

| 报告结论 | 查询证据 |
|---|---|
| 母体和冻结VIP | 第3节 snapshot CTE |
| 两活动严格漏斗和各VIP行为量 | 第4节漏斗+行为聚合 |

| 上线前后证据冲突 | 第10节同群窗口 |

复现规则：

1. 每条SQL必须限制 `$part_date` 和精确UTC0事件窗口。
2. VIP只使用活动开始快照，不使用窗口内MAX。
3. 配置门槛按Spin当时Level/VIP判断，冻结VIP仅用于分析分层。
4. 全项目RTP先按PriceCheatSheet归一化，再汇总；不能跨层直接相加金币。
5. 免费局派彩计入分子，免费局0下注不进入分母；同时单列免费局派彩。
6. 掉落率只能使用唯一Spin关联键；积分总量/Spin数是积分效率，不是命中率。
7. 所有用户级中间表只留在SQL CTE；文件只保存最终聚合。
8. `test_uid`、商城事件和reward数组限制解除后，应原SQL重跑并更新限制，不得猜测结果。
